## 4. CAN指纹采集脚本

```python
#!/usr/bin/env python3
# sagitar_fingerprint_collector.py

import json
from collections import defaultdict

def collect_sagitar_fingerprint():
    """
    采集大众速腾CAN指纹的示例脚本
    需要连接到车辆CAN总线
    """
    
    # 示例指纹数据结构（需要从实车采集）
    fingerprint_data = {
        # 基础车辆信息
        "0x1A0": 8,  # Bremse_1 - 制动信息
        "0x86": 8,   # LWI_01 - 转向角度
        "0xB2": 8,   # ESP_19 - ESP状态
        "0xFD": 8,   # ESP_21 - ESP数据
        
        # 仪表盘和车身信息
        "0x6B8": 8,  # Kombi_03 - 仪表盘
        "0x30F": 8,  # SWA_01 - 方向盘按键
        
        # 传动系统
        "0xAD": 8,   # Getriebe_11 - 变速箱（自动挡）
        "0x187": 8,  # Motor_EV_01 - 电机（混动/电动）
        
        # 其他重要消息
        "0x40": 8,   # Airbag_01 - 安全气囊
        "0x126": 8,  # HCA_01 - 车道保持（如果有）
        
        # 需要根据实车添加更多消息
    }
    
    return fingerprint_data

def save_fingerprint(data, filename="sagitar_fingerprint.json"):
    """保存指纹数据到文件"""
    with open(filename, 'w') as f:
        json.dump(data, f, indent=2)
    print(f"指纹数据已保存到 {filename}")

if __name__ == "__main__":
    # 采集指纹数据
    fingerprint = collect_sagitar_fingerprint()
    
    # 保存到文件
    save_fingerprint(fingerprint)
    
    print("大众速腾CAN指纹采集完成")
    print("请将此文件用于车型适配开发")
```
