# ===== raylib deps =====

依赖
RUN apt-get update && apt-get install -y \
  build-essential g++ cmake git \
  libx11-dev libxrandr-dev libxinerama-dev \
  libxcursor-dev libxi-dev libxext-dev \
  libxrender-dev libxfixes-dev libxkbcommon-dev \
  libgl1-mesa-dev libegl1-mesa-dev \
  && rm -rf /var/lib/apt/lists/*

编译
RUN git clone https://github.com/raysan5/raylib.git /tmp/raylib && \
    cd /tmp/raylib && git checkout 4.5.0 && \
    mkdir build && cd build && \
    cmake .. -DBUILD_SHARED_LIBS=OFF -DPLATFORM=Desktop && \
    make -j$(nproc)

复制到 op
RUN mkdir -p third_party/raylib/aarch64/include && \
cp /tmp/raylib/build/raylib/libraylib.a \
    third_party/raylib/aarch64/ && \
cp -r /tmp/raylib/src \
    third_party/raylib/aarch64/include/raylib

## 开机自启动
sudo nano /etc/systemd/system/openpilot.service


[Unit]
Description=OpenPilot Docker (Jetson)
After=docker.service graphical.target
Requires=docker.service

[Service]
Restart=always
RestartSec=5
TimeoutStartSec=0

# X11 显示（根据你现在的环境）
Environment=DISPLAY=:1
Environment=XDG_RUNTIME_DIR=/tmp/runtime-root

ExecStartPre=/usr/bin/xhost +local:docker

ExecStart=/usr/bin/docker run --rm \
  --name openpilot \
  --network host \
  --ipc host \
  --runtime nvidia \
  --privileged \
  -e DISPLAY=:1 \
  -e XDG_RUNTIME_DIR=/tmp/runtime-root \
  -v /tmp/.X11-unix:/tmp/.X11-unix:rw \
  -v /run/dbus/system_bus_socket:/run/dbus/system_bus_socket \
  -v /etc/machine-id:/etc/machine-id:ro \
  -v /data:/data \
  --device /dev/dri \
  --group-add video \
  mindcont/openpilot-arm64:latest \
  /home/batman/openpilot/selfdrive/ui/ui

ExecStop=/usr/bin/docker stop openpilot

[Install]
WantedBy=graphical.target


## 启用并启动服务
sudo systemctl daemon-reload
sudo systemctl enable openpilot
sudo systemctl start openpilot
